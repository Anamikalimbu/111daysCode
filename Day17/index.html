function fetchWithPromise() {
  const id = randomId();

  fetch(`https://jsonplaceholder.typicode.com/posts/${id}`)
    .then(res => {
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      return res.json();
    })
    .then(data => displayResult("promise", data))
    .catch(err => displayError("promise", err));
}
async function fetchWithAsync() {
  const id = randomId();
  try {
    const res  = await fetch(`…/posts/${id}`);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = await res.json();
    displayResult("async", data);
  } catch (err) {
    displayError("async", err);
  }
}
async function fetchParallel() {
  try {
    const [postRes, userRes] = await Promise.all([
      fetch("…/posts/1"),
      fetch("…/users/1")
    ]);
    const [post, user] = await Promise.all([
      postRes.json(),
      userRes.json()
    ]);
    displayResult("parallel", { post, user });
  } catch (err) {
    displayError("parallel", err);
  }
}
async function fetchWithError() {
  try {
    const res = await fetch("…/posts/99999"); // 404
    if (!res.ok)
      throw new Error(`Not Found — HTTP ${res.status}`);
    const data = await res.json();
    displayResult("error", data);
  } catch (err) {
    displayError("error", err); // caught here ✓
  }
}
